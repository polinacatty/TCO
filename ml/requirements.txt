# HTTP client.
# We use curl_cffi (not httpx / requests) because fedstat.ru's WAF
# performs TLS fingerprinting (JA3): pure-Python TLS stacks are blocked
# with HTTP 403 regardless of headers.  curl_cffi delegates the handshake
# to libcurl with Chrome's exact cipher suites / ALPN order, which gets
# us past the WAF.  See docs/03_data_collection/03_fuel_data_recon.md.
curl_cffi>=0.7,<1.0

# httpx is kept for non-fedstat sources (CBR, plain Rosstat XLSX, etc.)
# where pure-Python TLS works fine.
httpx>=0.26,<1.0
